%==============================================================================
%  PACKAGES (extra to those loaded by ozu-thesis.cls)
%  The class already loads: report base, times font, calc, tocloft.
%  Margins, line spacing, headers, title/approval pages are handled by the class.
%==============================================================================

% ---- Encoding (so UTF-8 in references.bib renders correctly) ----------------
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}

% ---- Math -------------------------------------------------------------------
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{mathtools}

% ---- Figures & tables -------------------------------------------------------
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{float}          % [H] placement
\usepackage{booktabs}       % \toprule, \midrule, \bottomrule
\usepackage{multirow}
\usepackage{subcaption}

% ---- Units ------------------------------------------------------------------
\usepackage{siunitx}
\sisetup{per-mode=symbol, detect-all}

% ---- Acronyms (List of Acronyms and Abbreviations) --------------------------
% No "printonlyused": the list prints every entry defined in frontmatter/acronyms.
\usepackage{acronym}

% ---- Captions (OzU template style) ------------------------------------------
\usepackage[justification=justified,singlelinecheck=false]{caption}
\captionsetup{skip=6pt}

% ---- Misc -------------------------------------------------------------------
\usepackage{xcolor}
\usepackage{csquotes}
\usepackage{url}
\usepackage{indentfirst}

% ---- Links & smart cross-references -----------------------------------------
% hyperref must precede cleveref; the class is given the 'hyperref' option.
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=black,
  citecolor=black,
  urlcolor=blue,
}
\usepackage[capitalize,noabbrev]{cleveref}

% Paragraph indent per OzU template (1 tab = 1.27 cm).
\setlength{\parindent}{1.27cm}
